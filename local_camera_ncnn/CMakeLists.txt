cmake_minimum_required(VERSION 3.24)
project(local_camera_ncnn LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)

# CPU-only ncnn. No MediaPipe, TensorFlow, Bazel or model conversion at runtime.
set(NCNN_VULKAN OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_BENCHMARK OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(NCNN_RUNTIME_CPU ON CACHE BOOL "" FORCE)
# Keep runtime SSE/AVX2 dispatch; omit the larger AVX-512 variants.
set(NCNN_AVX512 OFF CACHE BOOL "" FORCE)
option(USE_SYSTEM_NCNN "Use an already installed ncnn" OFF)
if(USE_SYSTEM_NCNN)
  find_package(ncnn CONFIG REQUIRED)
else()
  FetchContent_Declare(ncnn URL https://github.com/Tencent/ncnn/archive/refs/tags/20250503.zip
    URL_HASH SHA256=d2b6250aedbd9014fe3e03d7206758e99c6337d73c5b27b24bcbf64220dc9d26)
  FetchContent_MakeAvailable(ncnn)
endif()

option(BUILD_BUNDLED_OPENCV "Build just the required OpenCV modules" OFF)
if(BUILD_BUNDLED_OPENCV)
  set(BUILD_LIST core,imgproc,imgcodecs,highgui,videoio CACHE STRING "" FORCE)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(BUILD_WITH_STATIC_CRT OFF CACHE BOOL "" FORCE)
  set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(BUILD_PERF_TESTS OFF CACHE BOOL "" FORCE)
  set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(BUILD_opencv_apps OFF CACHE BOOL "" FORCE)
  set(BUILD_JAVA OFF CACHE BOOL "" FORCE)
  # OpenCV 4.10 declares this version as a BOOL option; avoid an invalid "OFF"
  # argument when the Python interpreter is unavailable (we build no bindings).
  set(OPENCV_PYTHON3_VERSION "3" CACHE STRING "" FORCE)
  set(WITH_IPP OFF CACHE BOOL "" FORCE)
  set(WITH_ITT OFF CACHE BOOL "" FORCE)
  set(WITH_OPENCL OFF CACHE BOOL "" FORCE)
  set(WITH_FFMPEG OFF CACHE BOOL "" FORCE)
  set(WITH_MSMF OFF CACHE BOOL "" FORCE)
  set(WITH_DSHOW ON CACHE BOOL "" FORCE)
  set(WITH_WEBP OFF CACHE BOOL "" FORCE)
  set(WITH_TIFF OFF CACHE BOOL "" FORCE)
  set(WITH_OPENEXR OFF CACHE BOOL "" FORCE)
  set(WITH_JASPER OFF CACHE BOOL "" FORCE)
  set(WITH_OPENJPEG OFF CACHE BOOL "" FORCE)
  set(OPENCV_ENABLE_ALLOCATOR_STATS OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(opencv URL https://github.com/opencv/opencv/archive/refs/tags/4.10.0.zip
    URL_HASH SHA256=3810bca2b1d1c572912df0ac3888126341f3762dfd28e91068c805fb656d0e51)
  FetchContent_MakeAvailable(opencv)
  set(OpenCV_LIBS opencv_core opencv_imgproc opencv_imgcodecs opencv_highgui opencv_videoio)
  set(OpenCV_INCLUDE_DIRS "${opencv_BINARY_DIR}" "${CMAKE_BINARY_DIR}")
  foreach(module core imgproc imgcodecs highgui videoio)
    list(APPEND OpenCV_INCLUDE_DIRS "${opencv_SOURCE_DIR}/modules/${module}/include")
  endforeach()
else()
  find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs highgui videoio)
endif()

add_executable(local_camera local_camera.cpp)
target_include_directories(local_camera PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(local_camera PRIVATE ncnn ${OpenCV_LIBS})
if(MSVC)
  target_compile_options(local_camera PRIVATE /W4 /utf-8)
else()
  target_compile_options(local_camera PRIVATE -Wall -Wextra)
endif()
set_target_properties(local_camera PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(config Release Debug RelWithDebInfo MinSizeRel)
  string(TOUPPER "${config}" upper)
  set_target_properties(local_camera PROPERTIES "RUNTIME_OUTPUT_DIRECTORY_${upper}" "${CMAKE_BINARY_DIR}/bin")
endforeach()
add_custom_target(local_camera_models
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/models" "$<TARGET_FILE_DIR:local_camera>/models")
add_dependencies(local_camera local_camera_models)
enable_testing()
add_test(NAME model_smoke COMMAND local_camera --check-model)
add_test(NAME hand_image COMMAND local_camera --image "${CMAKE_CURRENT_SOURCE_DIR}/testdata/hand.jpg" --expect-hands 1)
add_test(NAME invalid_fps COMMAND local_camera --fps 0)
set_tests_properties(invalid_fps PROPERTIES WILL_FAIL TRUE)
