cmake_minimum_required(VERSION 3.24)
project(local_camera_ncnn_pi LANGUAGES C CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Build this project on Raspberry Pi OS 64-bit (Linux).")
endif()
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
  message(FATAL_ERROR "A 64-bit ARM toolchain is required (aarch64).")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Use the distribution's OpenCV; only ncnn is compiled from source.
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs videoio)
find_package(Threads REQUIRED)
option(USE_SYSTEM_NCNN "Use a preinstalled ncnn via ncnn_DIR" OFF)
if(USE_SYSTEM_NCNN)
  find_package(ncnn CONFIG REQUIRED)
else()
  include(FetchContent)
  set(NCNN_VULKAN OFF CACHE BOOL "" FORCE)
  set(NCNN_OPENMP ON CACHE BOOL "" FORCE)
  set(NCNN_RUNTIME_CPU ON CACHE BOOL "" FORCE)
  set(NCNN_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(NCNN_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
  set(NCNN_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(NCNN_BUILD_BENCHMARK OFF CACHE BOOL "" FORCE)
  set(NCNN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(ncnn
    URL https://github.com/Tencent/ncnn/archive/refs/tags/20260526.zip
    URL_HASH SHA256=5aedf25e22d4a9ab7785d33774affee96b55b546e85318cc351a48a365d15ced)
  FetchContent_MakeAvailable(ncnn)
endif()

add_executable(local_camera local_camera.cpp)
target_include_directories(local_camera PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(local_camera PRIVATE ncnn ${OpenCV_LIBS} Threads::Threads)
target_compile_options(local_camera PRIVATE -Wall -Wextra)
set_target_properties(local_camera PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_custom_target(local_camera_models
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/models" "$<TARGET_FILE_DIR:local_camera>/models")
add_dependencies(local_camera local_camera_models)
add_executable(ncnn_probe EXCLUDE_FROM_ALL conversion/ncnn_probe.cpp)
target_link_libraries(ncnn_probe PRIVATE ncnn)
if(USE_SYSTEM_NCNN)
  target_compile_definitions(ncnn_probe PRIVATE NCNN_RELEASE_TAG="system-package")
else()
  target_compile_definitions(ncnn_probe PRIVATE NCNN_RELEASE_TAG="20260526")
endif()

enable_testing()
add_executable(uv_sender_test tests/uv_sender_test.cpp)
target_include_directories(uv_sender_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
add_test(NAME uv_protocol COMMAND uv_sender_test)
add_test(NAME model_smoke COMMAND local_camera --check-model)
add_test(NAME hand_image COMMAND local_camera --image "${CMAKE_CURRENT_SOURCE_DIR}/testdata/hand.jpg" --expect-hands 1)
add_test(NAME invalid_fps COMMAND local_camera --fps 0)
set_tests_properties(invalid_fps PROPERTIES WILL_FAIL TRUE)
